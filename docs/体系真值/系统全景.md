## 4. 复现命令（现算协议覆盖）

```bash
cd <项目目录>
python - <<'PY'
import re
srv = open("服务端/runtime/tools/fake_flash_server.py", encoding="utf-8").read()
m = re.search(r"class ClientOpcode\b.*?(?=\nclass )", srv, re.S)
srv_ops = {int(v): n for n, v in re.findall(r'^\s+([A-Z][A-Z0-9_]+)\s*=\s*(\d+)\s*$', m.group(0), re.M)}
be = open("脚本/单例/后端连接.gd", encoding="utf-8").read()
cli = {int(v) for _, v in re.findall(r'^const\s+(OP_[A-Z0-9_]+)\s*:=\s*(\d+)\s*$', be, re.M)}
print("覆盖 %d/%d" % (len(set(srv_ops) & cli), len(srv_ops)))
PY
# 原版窗口类清单
grep -rlE "class \w+ extends (MWindow|UIWindow)" ../2.2.3/map_extract/out/mainlib_as/scripts | wc -l
```
